cat > /etc/frr/frr.conf <<'EOF'
interface gre
 no ip ospf passive

exit
!

!

EOF

    chain postrouting {
        type nat hook postrouting priority srcnat;
        oifname "enp7s1" masquerade
    }
}
EOF
